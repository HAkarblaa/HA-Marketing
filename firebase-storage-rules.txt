rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /ha_marketplace/{userId}/{itemId}/{fileName} {
      allow read: if true;
      allow create: if request.auth != null && request.auth.uid == userId && request.resource.size < 2 * 1024 * 1024 && request.resource.contentType == 'image/jpeg';
      allow delete: if request.auth != null && request.auth.uid == userId;
      allow update: if false;
    }
  }
}
